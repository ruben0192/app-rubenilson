const express = require('express');
const app = express();
app.use(express.json());

let users = [
  { id: 1, nome: 'João' },
  { id: 2, nome: 'Maria' }
];

app.get('/usuarios', (req, res) => {
  res.json(users);
});

app.post('/usuarios', (req, res) => {
  const { nome } = req.body;
  if (!nome) return res.status(400)
    .json({ error: 'Nome é obrigatório' });

  const newUser = { id: users.length + 1, nome };
  users.push(newUser);
  res.status(201).json(newUser);
});

app.listen(3000);
